cmake_minimum_required(VERSION 3.24)
project(ElunaText LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(Eluna.Text SHARED Emoji.cpp)
target_compile_features(Eluna.Text PRIVATE cxx_std_17)
target_compile_definitions(Eluna.Text PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(Eluna.Text PRIVATE /W4 /WX /utf-8 /guard:cf)
target_link_options(Eluna.Text PRIVATE /DYNAMICBASE /NXCOMPAT /guard:cf)
target_link_libraries(Eluna.Text PRIVATE d2d1 dwrite windowscodecs ole32)
