$ErrorActionPreference = 'Stop'
$taskRoot = Split-Path -Parent $PSScriptRoot
$taskExe = Join-Path $taskRoot 'artifacts/social-portable-ready/Eluna.exe'
$taskRegression = Join-Path $taskRoot 'artifacts/social-regression-ready'
New-Item -ItemType Directory -Force $taskRegression | Out-Null
$env:DOTNET_ROOT = Join-Path $taskRegression 'absent-runtime'
$env:DOTNET_ROOT_X64 = $env:DOTNET_ROOT
$env:DOTNET_MULTILEVEL_LOOKUP = '0'
$env:DOTNET_BUNDLE_EXTRACT_BASE_DIR = Join-Path $taskRegression 'extracted'
foreach ($taskKind in @('discord','preview','ui-ux','voice','agents')) {
    $taskOutput = Join-Path $taskRegression $taskKind
    if ($taskKind -in @('preview','ui-ux')) { $taskArgs = @('--capture-dir', ('"'+$taskOutput+'"')); if ($taskKind -eq 'ui-ux') { $taskArgs += '--verify-ui-ux' } }
    else { $taskArgs = @('--verify-'+$taskKind, ('"'+$taskOutput+'"'), '--data-dir', ('"'+(Join-Path $taskOutput 'data')+'"')) }
    $taskProcess = Start-Process -FilePath $taskExe -ArgumentList $taskArgs -WindowStyle Hidden -PassThru
    $taskDeadline = [DateTime]::UtcNow.AddMinutes(5)
    while (-not $taskProcess.WaitForExit(1000)) { if ([DateTime]::UtcNow -gt $taskDeadline) { $taskProcess.Kill(); throw "$taskKind timed out" } }
    $taskResult = Join-Path $taskOutput 'result.txt'
    if (Test-Path -LiteralPath $taskResult) { Get-Content -LiteralPath $taskResult }
    if ($taskProcess.ExitCode -ne 0 -or -not (Test-Path -LiteralPath $taskResult) -or (Get-Content -LiteralPath $taskResult -Raw) -notmatch ': OK') { throw "$taskKind failed" }
}
