$ErrorActionPreference = 'Stop'
$taskRoot = Split-Path -Parent $PSScriptRoot
$portable = Join-Path $PSScriptRoot 'agents-portable-confirmed'
New-Item -ItemType Directory -Force $portable | Out-Null
Copy-Item -LiteralPath (Join-Path $taskRoot 'artifacts/release-agents/Eluna.exe') -Destination (Join-Path $portable 'Eluna.exe') -Force
$env:DOTNET_ROOT = Join-Path $portable 'absent-runtime'
$env:DOTNET_ROOT_X64 = $env:DOTNET_ROOT
$env:DOTNET_MULTILEVEL_LOOKUP = '0'
$env:DOTNET_BUNDLE_EXTRACT_BASE_DIR = Join-Path $portable 'extracted'
$taskModel = Join-Path $taskRoot 'artifacts/media-integration-4b/Models/db5d6f08d43ef2e78a24a8f60ec7133c2dc7c933ac33e53f73df1206a8b9a30d/model.json'
$taskOutput = Join-Path $portable 'live'
$taskProcess = Start-Process -FilePath (Join-Path $portable 'Eluna.exe') -ArgumentList @('--verify-agents', ('"'+$taskOutput+'"'), '--data-dir', ('"'+(Join-Path $taskOutput 'data')+'"'), '--model-meta', ('"'+$taskModel+'"')) -WindowStyle Hidden -PassThru
$taskDeadline = [DateTime]::UtcNow.AddMinutes(10)
$taskLastStage = ''
while (-not $taskProcess.WaitForExit(1000)) {
    if ([DateTime]::UtcNow -gt $taskDeadline) { $taskProcess.Kill(); throw 'Agent fixture timed out' }
    $taskProgress = Join-Path $taskOutput 'progress.json'
    if (Test-Path -LiteralPath $taskProgress) {
        try { $taskStage = (Get-Content -LiteralPath $taskProgress -Raw | ConvertFrom-Json).stage; if ($taskStage -ne $taskLastStage) { Write-Output $taskStage; $taskLastStage = $taskStage } } catch { }
    }
}
Get-Content -LiteralPath (Join-Path $taskOutput 'result.txt')
if ($taskProcess.ExitCode -ne 0) { throw 'Agent fixture failed' }
$taskRuntime = Get-ChildItem -LiteralPath (Join-Path $taskOutput 'data/Runtime') -Directory | Select-Object -First 1
if ($null -eq $taskRuntime) { throw 'Bundled native engines were not extracted' }
$taskNative = foreach ($backend in @('cpu', 'vulkan')) {
    foreach ($name in @('llama-server.exe', 'sd-cli.exe')) {
        $binary = Join-Path $taskRuntime.FullName "$backend/$name"
        $lines = & $binary --version 2>&1
        if ($LASTEXITCODE -ne 0) { throw "$backend/$name failed" }
        [pscustomobject]@{ Backend = $backend; Name = $name; Sha256 = (Get-FileHash -LiteralPath $binary -Algorithm SHA256).Hash; Output = ($lines -join "`n") }
    }
}
$taskNative | ConvertTo-Json -Depth 5 | Set-Content -LiteralPath (Join-Path $portable 'native.json') -Encoding utf8
Write-Output 'Standalone agents and four native engines: OK'
