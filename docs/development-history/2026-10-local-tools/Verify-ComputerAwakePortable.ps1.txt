$ErrorActionPreference = 'Stop'
$taskRoot = Split-Path -Parent $PSScriptRoot
$portable = Join-Path $PSScriptRoot 'computer-awake-portable'
New-Item -ItemType Directory -Force $portable | Out-Null
Copy-Item -LiteralPath (Join-Path $taskRoot 'artifacts/release-computer-awake/Eluna.exe') -Destination (Join-Path $portable 'Eluna.exe') -Force
$env:DOTNET_ROOT = Join-Path $portable 'absent-runtime'
$env:DOTNET_ROOT_X64 = $env:DOTNET_ROOT
$env:DOTNET_MULTILEVEL_LOOKUP = '0'
$env:DOTNET_BUNDLE_EXTRACT_BASE_DIR = Join-Path $portable 'extracted'
$preview = Join-Path $portable 'ui'
$process = Start-Process -FilePath (Join-Path $portable 'Eluna.exe') -ArgumentList @('--capture-dir', $preview, '--verify-ui-ux') -WindowStyle Hidden -PassThru
if (-not $process.WaitForExit(90000)) { $process.Kill(); throw 'Standalone UI check timed out.' }
if ($process.ExitCode -ne 0 -or -not (Test-Path (Join-Path $preview 'verification.json'))) { throw 'Standalone UI check failed.' }
$report = Get-Content -LiteralPath (Join-Path $preview 'verification.json') -Raw | ConvertFrom-Json
if (-not $report.computerUseKeepAwake) { throw 'Standalone UI did not verify the power setting.' }
$runtime = Get-ChildItem -LiteralPath (Join-Path $preview 'preview-data/Runtime') -Directory | Select-Object -First 1
if ($null -eq $runtime) { throw 'Bundled engines were not extracted.' }
$results = foreach ($backend in @('cpu', 'vulkan')) {
    foreach ($name in @('llama-server.exe', 'sd-cli.exe')) {
        $binary = Join-Path $runtime.FullName "$backend/$name"
        $lines = & $binary --version 2>&1
        if ($LASTEXITCODE -ne 0) { throw "$backend/$name failed: $lines" }
        [pscustomobject]@{ Backend = $backend; Name = $name; ExitCode = $LASTEXITCODE; Sha256 = (Get-FileHash -LiteralPath $binary -Algorithm SHA256).Hash; Output = ($lines -join "`n") }
    }
}
$results | ConvertTo-Json -Depth 5 | Set-Content -LiteralPath (Join-Path $portable 'native.json') -Encoding utf8
Get-FileHash -LiteralPath (Join-Path $portable 'Eluna.exe') -Algorithm SHA256 | ConvertTo-Json | Set-Content -LiteralPath (Join-Path $portable 'exe.json') -Encoding utf8
Write-Output 'Standalone UI, power settings persistence/rendering and all four bundled engines: OK'
